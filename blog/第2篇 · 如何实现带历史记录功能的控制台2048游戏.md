# 第2篇 · 如何实现带历史记录功能的控制台2048游戏

第 1 篇把 2048 的规则实现为一个可运行的控制台程序，但它有一个共同的局限：**进程退出后状态不保留**。关闭窗口后，上一局的分数、棋盘以及此前的每一步操作都会消失，只能重新开始。

历史记录为游戏提供两套机制：**逐步轨迹 + 状态快照**。逐步轨迹负责将每一局、每一步（方向、棋盘、得分）持久化，随时可回看；状态快照负责在任意时刻冻结棋盘，下次启动时继续。

该功能有几种实现选择：存入数据库最为规范，但对单机小游戏属于明显的过度设计；仅使用单个整体 JSON 文件最为简单，却要在每步重写全量数据，且写入中途崩溃会破坏整个存档。此处采用折中方案：**JSON 存储续玩快照 + CSV 追加逐步轨迹**，并配合原子写入与损坏回退。

本文将从**历史记录数据结构、状态保存与加载、历史查看与管理、退出时的数据更新、启动续玩**五个方面，系统讲解如何给控制台 2048 加上可靠的历史记录功能。

## 1 游戏历史记录管理

为便于玩家查看并管理自己的游戏历史，程序将每局游戏的相关数据（例如分数、网格状态等）保存到本地文件。历史记录包含每局游戏的操作步骤、分数、移动后的网格状态等。

**关键点**：

- **游戏ID生成**：每局游戏会生成一个唯一的 ID，基于当前的时间戳。
- **CSV 文件记录**：每一步游戏操作（包括方块的移动、得分和网格变化）会记录到 CSV 文件中。
- **JSON 文件记录**：游戏的完整状态（例如：网格、分数、胜利状态等）会保存到 JSON 文件中，以供后续恢复。

**代码实现**：

```python
import random
import os
import msvcrt  # Windows下的键盘输入处理
import csv
import json
import datetime

class Console2048WithHistory:
    def __init__(self, grid_size=4):
        # 初始化游戏设置
        self.grid_size = grid_size
        self.grid = [[0] * grid_size for _ in range(grid_size)]
        self.score = 0
        self.best_score = 0
        self.game_over = False
        self.won = False

        # 历史记录相关
        self.history_dir = os.path.join(os.path.dirname(os.path.abspath(__file__)), "game_history")
        self.history_file = None
        self.game_id = None
        self.game_start_time = None
        self.current_move = 0

        if not os.path.exists(self.history_dir):
            os.makedirs(self.history_dir)

        # 读取历史最高分
        self.load_best_score()

    def load_best_score(self):
        """加载最高分"""
        best_score_file = os.path.join(self.history_dir, "best_score.json")
        if os.path.exists(best_score_file):
            try:
                with open(best_score_file, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    self.best_score = data.get('best_score', 0)
            except (json.JSONDecodeError, IOError) as e:
                # 记录错误但继续运行
                pass

    def save_best_score(self):
        """保存最高分"""
        best_score_file = os.path.join(self.history_dir, "best_score.json")
        try:
            with open(best_score_file, 'w', encoding='utf-8') as f:
                json.dump({'best_score': self.best_score}, f, ensure_ascii=False)
        except IOError as e:
            # 记录错误但继续运行
            pass

    def start_new_game(self):
        """开始新游戏"""
        # 生成游戏ID（基于时间戳）
        self.game_id = datetime.datetime.now().strftime("%Y%m%d_%H%M%S_%f")
        # 记录游戏开始时间
        self.game_start_time = datetime.datetime.now()

        # 创建游戏历史记录文件
        game_dir = os.path.join(self.history_dir, self.game_id)
        os.makedirs(game_dir, exist_ok=True)

        # 创建历史记录CSV文件
        self.history_file = os.path.join(game_dir, "history.csv")
        with open(self.history_file, 'w', newline='', encoding='utf-8') as f:
            writer = csv.writer(f)
            writer.writerow(["步数", "操作类型", "操作得分", "总分", "移动后的网格状态", "随机添加后的网格状态"])

        # 初始化游戏状态
        self.grid = [[0] * self.grid_size for _ in range(self.grid_size)]
        self.score = 0
        self.game_over = False
        self.won = False
        self.current_move = 0

        # 添加两个初始方块
        for _ in range(2):
            self.add_new_tile()

        # 记录初始状态
        self.record_move("初始状态", 0, self.grid, self.grid)

        # 更新目录文档
        self.update_index()

    def record_move(self, move_type, move_score, after_move_grid, after_add_tile_grid):
        """记录一步操作"""
        if not self.history_file:
            return

        # 增加步数
        self.current_move += 1

        # 记录到CSV文件
        with open(self.history_file, 'a', newline='', encoding='utf-8') as f:
            writer = csv.writer(f)
            # 将网格转换为JSON字符串保存
            after_move_grid_json = json.dumps(after_move_grid, ensure_ascii=False)
            after_add_tile_grid_json = json.dumps(after_add_tile_grid, ensure_ascii=False)
            writer.writerow([self.current_move, move_type, move_score, self.score, after_move_grid_json, after_add_tile_grid_json])

        # 保存游戏状态用于恢复
        self.save_game_state()
```

至此，历史记录的写入侧已经完整：`start_new_game()` 负责创建目录与表头，`record_move()` 负责逐步追加 CSV，并同时保存最新状态。

## 2 游戏状态保存与加载

为增强用户体验，程序增加了游戏状态的保存与加载功能。每次玩家执行操作后，游戏的网格状态、分数等信息会保存为 JSON 文件。因此在玩家退出游戏后，下一次启动时仍可继续之前的游戏。

**关键点**：

- **状态保存**：每执行一步操作后，游戏状态（如网格、分数等）会保存在 JSON 文件中。
- **状态恢复**：游戏支持恢复最近未完成的游戏，通过加载保存的状态继续游戏。

**代码实现**：

```python
def save_game_state(self):
    """保存游戏状态"""
    if not self.game_id:
        return

    game_dir = os.path.join(self.history_dir, self.game_id)
    state_file = os.path.join(game_dir, "game_state.json")

    game_state = {
        "grid": self.grid,
        "score": self.score,
        "best_score": self.best_score,
        "game_over": self.game_over,
        "won": self.won,
        "current_move": self.current_move,
        "last_updated": datetime.datetime.now().isoformat()
    }

    try:
        with open(state_file, 'w', encoding='utf-8') as f:
            json.dump(game_state, f, ensure_ascii=False, indent=2)
    except IOError as e:
        pass

def load_game_state(self, game_id):
    """加载指定游戏ID的状态"""
    game_dir = os.path.join(self.history_dir, game_id)
    state_file = os.path.join(game_dir, "game_state.json")

    if not os.path.exists(state_file):
        return False

    try:
        with open(state_file, 'r', encoding='utf-8') as f:
            game_state = json.load(f)

        self.game_id = game_id
        self.grid = game_state.get('grid', [[0] * self.grid_size for _ in range(self.grid_size)])
        self.score = game_state.get('score', 0)
        self.best_score = game_state.get('best_score', 0)
        self.game_over = game_state.get('game_over', False)
        self.won = game_state.get('won', False)
        self.current_move = game_state.get('current_move', 0)

        # 更新历史文件路径
        self.history_file = os.path.join(game_dir, "history.csv")

        # 从索引中获取游戏开始时间
        index_file = os.path.join(self.history_dir, "index.json")
        if os.path.exists(index_file):
            with open(index_file, 'r', encoding='utf-8') as f:
                index_data = json.load(f)
                for game in index_data:
                    if game.get('game_id') == game_id:
                        start_time_str = game.get('start_time', datetime.datetime.now().isoformat())
                        self.game_start_time = datetime.datetime.fromisoformat(start_time_str)
                        break

        if not self.game_start_time:
            self.game_start_time = datetime.datetime.now()

        return True
    except (json.JSONDecodeError, IOError, ValueError) as e:
        return False
```

## 3 游戏历史记录查看与管理

程序通过读取 `index.json` 文件来查看最近未完成的游戏，玩家可以选择加载最新的未完成游戏，从而继续上一次的对局，无需从头开始。

**关键点**：

- **对局索引**：`index.json` 记录每一局的概要（game_id、开始时间、分数、是否完成、最后更新时间），据此可以选出最近一局未完成的对局。
- **加载历史游戏**：玩家可以选择继续最近的一局未完成的游戏，恢复之前的网格状态和分数。
- **放弃续玩**：如果玩家不想继续，则把这一局在索引里标记为已完成，避免下次启动时又被当作可续玩的"活档"。

**代码实现**：

```python
def get_last_unfinished_game(self):
    """获取最后一个未完成且非游戏结束状态的游戏信息（活档）"""
    index_file = os.path.join(self.history_dir, "index.json")
    if not os.path.exists(index_file):
        return None

    try:
        with open(index_file, 'r', encoding='utf-8') as f:
            index_data = json.load(f)

        # 筛选真正的活档：未完成的游戏
        unfinished_games = [game for game in index_data if not game.get('is_completed', False)]
        if not unfinished_games:
            return None

        # 按最后更新时间排序，获取最新的未完成游戏
        unfinished_games.sort(key=lambda x: x.get('last_updated', ''), reverse=True)
        return unfinished_games[0]
    except (json.JSONDecodeError, IOError) as e:
        return None

def mark_game_completed(self, game_id):
    """把指定对局在索引中标记为已完成（用于用户放弃续玩时结算旧档）"""
    if not game_id:
        return
    index_file = os.path.join(self.history_dir, "index.json")
    if not os.path.exists(index_file):
        return
    try:
        with open(index_file, 'r', encoding='utf-8') as f:
            index_data = json.load(f)
        for info in index_data:
            if info.get('game_id') == game_id:
                info['is_completed'] = True
                info['last_updated'] = datetime.datetime.now().isoformat()
                break
        with open(index_file, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
    except (json.JSONDecodeError, IOError) as e:
        # 记录错误但继续运行
        pass
```

## 4 游戏退出与数据更新

当玩家退出游戏时，程序会更新游戏索引，确保当前数据被及时保存。退出时，游戏状态会被标记为未完成，玩家可以随时恢复上次的状态。

**关键点**：

- **游戏结束时更新索引**：每当一局游戏结束时，游戏的结果会更新到 `index.json` 文件中。
- **游戏退出时保存未完成状态**：当玩家退出时，游戏状态会被标记为未完成，以便玩家以后继续。

**代码实现**：

```python
def update_index(self, is_completed=None):
    """更新目录文档"""
    if not self.game_id or not self.game_start_time:
        return

    # 刷新并持久化历史最高分
    if self.score > self.best_score:
        self.best_score = self.score
        self.save_best_score()

    index_file = os.path.join(self.history_dir, "index.json")

    # 读取现有索引
    index_data = []
    if os.path.exists(index_file):
        try:
            with open(index_file, 'r', encoding='utf-8') as f:
                index_data = json.load(f)
        except (json.JSONDecodeError, IOError) as e:
            index_data = []

    completed_status = is_completed if is_completed is not None else (self.game_over or self.won)

    # 创建当前游戏的索引条目
    game_info = {
        "game_id": self.game_id,
        "start_time": self.game_start_time.isoformat(),
        "score": self.score,
        "best_score": self.best_score,
        "is_completed": completed_status,
        "last_updated": datetime.datetime.now().isoformat()
    }

    # 已存在则更新，不存在则追加
    for info in index_data:
        if info["game_id"] == self.game_id:
            info.update(game_info)
            break
    else:
        index_data.append(game_info)

    # 写回索引文件
    try:
        with open(index_file, 'w', encoding='utf-8') as f:
            json.dump(index_data, f, ensure_ascii=False, indent=2)
    except IOError as e:
        pass
```

## 5 启动续玩与主循环

上述能力由 `run()` 串联：启动时先检查是否存在未完成的对局，若有则询问是否继续（选择 `n` 时用 `mark_game_completed()` 结算旧档）；随后进入主循环，显示网格、响应按键，每次成功移动都记录一步并刷新索引；游戏结束或退出时再通过 `update_index()` 落盘最终状态。

**代码实现**：

```python
def run(self):
    """运行游戏主循环"""
    # 检查是否有未完成的游戏
    last_unfinished = self.get_last_unfinished_game()
    if last_unfinished:
        print(f"检测到未完成的游戏：")
        print(f"游戏ID: {last_unfinished.get('game_id')}")
        print(f"开始时间: {last_unfinished.get('start_time')}")
        print(f"当前分数: {last_unfinished.get('score')}")
        print(f"最高分: {last_unfinished.get('best_score')}")
        print(f"最后更新: {last_unfinished.get('last_updated')}")
        print()

        # 询问用户是否继续
        while True:
            print("是否继续上一把游戏？ (y/n): ")
            choice = msvcrt.getch().decode('utf-8', errors='ignore').lower()
            print(choice)
            if choice == 'y':
                # 尝试加载游戏状态
                if self.load_game_state(last_unfinished.get('game_id')):
                    print("游戏加载成功！")
                    input("按回车键继续...")
                    break
                else:
                    print("游戏加载失败，开始新游戏。")
                    self.start_new_game()
                    break
            elif choice == 'n':
                # 放弃续玩，把旧对局标记为已结束，避免下次启动再次被当作可续玩的活档
                self.mark_game_completed(last_unfinished.get('game_id'))
                self.start_new_game()  # 开始新游戏
                break
    else:
        self.start_new_game()  # 没有未完成游戏，直接开始新游戏

    while True:
        # 显示网格
        self.display_grid()  # 显示当前状态
        self.check_win()  # 检查是否赢了

        # 检查是否游戏结束
        if self.check_game_over():  # 如果游戏结束
            self.display_grid()  # 显示最终状态
            # 更新索引标记为已完成
            self.update_index(is_completed=True)
            input("按回车键重新开始...")
            self.start_new_game()  # 重置游戏
            continue

        # 获取用户输入
        key = self.get_key()
        if key == 'quit':  # 如果用户选择退出
            print("感谢游玩！再见！")
            # 退出时更新索引，标记为未完成
            self.update_index(is_completed=False)
            break

        if key in ('up', 'down', 'left', 'right'):  # 处理用户移动
            previous_score = self.score  # 记录移动前的分数
            if self._move(key):  # 如果移动成功
                after_move_grid = [row.copy() for row in self.grid]  # 移动后的网格（尚未添加新方块）
                self.add_new_tile()  # 添加新方块
                move_score = self.score - previous_score  # 计算本次移动获得的分数
                self.record_move(f"向{key}移动", move_score, after_move_grid, self.grid)
                # 更新索引文件中的分数信息
                self.update_index()
```

## 7 小结

这些新增功能提升了 **2048 游戏** 的交互性：每局对局的完整操作轨迹与状态都会落盘（CSV + JSON），下次启动时可以直接恢复未完成的对局，历史最高分也会单独持久化。

JSON + CSV 的文件方案适合单机存档、续玩与轨迹导出（CSV 可直接导入 pandas），也支撑了第 3 篇启动页的"最近一局"；多端同步、并发写入与海量检索很快会遇到瓶颈，届时需改用数据库。
